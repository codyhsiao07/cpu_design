# ========= 小工具：從某個 bd pin 找到它連到的 external bd port =========
proc _ext_port_from_pin {pin_obj} {
  set nets [get_bd_nets -quiet -of_objects $pin_obj]
  if {[llength $nets] == 0} { return "" }
  set ports [get_bd_ports -quiet -of_objects [lindex $nets 0]]
  if {[llength $ports] == 0} { return "" }
  return [lindex $ports 0]
}

# ========= 小工具：確保 pin 已 external，並回傳對應的 bd port（不會回傳空字串） =========
proc _ensure_pin_external {pin_path} {
  set pin_obj [get_bd_pins -quiet $pin_path]
  if {[llength $pin_obj] == 0} { error "Pin not found: $pin_path" }

  # 若已經 external 或已接到某個 bd port，就直接回傳 port
  set p [_ext_port_from_pin $pin_obj]
  if {$p ne ""} { return $p }

  # 否則嘗試 external
  catch { make_bd_pins_external $pin_obj } _

  # 再找一次
  set p [_ext_port_from_pin $pin_obj]
  if {$p eq ""} { error "Failed to externalize pin: $pin_path" }
  return $p
}

# ========= 小工具：確保 interface 已 external（已 external 就忽略） =========
proc _ensure_intf_external {intf_path} {
  set iobj [get_bd_intf_pins -quiet $intf_path]
  if {[llength $iobj] == 0} { error "Interface pin not found: $intf_path" }
  catch { make_bd_intf_pins_external $iobj } _
}
